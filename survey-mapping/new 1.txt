// ─────────────────────────────────────────────────────────────
//  WCAC2 DETAIL MAP

// 1) Initialize the detail map in the left pane
var mapWcac2 = L.map('map-wcac2-detail')
  .setView([36.853726, -107.804702], 12);  // center on LA MANGA
  
  // 2) Add lat/long tracker
mapWcac2.on('mousemove', function(e) {
  document.getElementById('latlong-wcac2').innerHTML =
    `Lat: ${e.latlng.lat.toFixed(5)} Long: ${e.latlng.lng.toFixed(5)}`;
	});

// 3) Add a scale bar
	L.control.scale({
		position: 'bottomright',
		imperial: true,
		metric: true,
		maxWidth: 200
	}).addTo(mapWcac2)	
	 

// 2) Define all the base layers (give them Wm suffix so nothing collides)
var stamenTerrainWc = L.tileLayer(
  'https://tiles.stadiamaps.com/tiles/stamen_terrain/{z}/{x}/{y}.jpg?api_key=a8a97fb6-0fdd-4f58-a626-8f8b8f7850d6', {
    attribution: 'Stamen Terrain', maxZoom: 18, tileSize: 512, zoomOffset: -1
  }
);
var esriWorldImageryWc = L.tileLayer(
  'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', {
    attribution: 'Tiles © Esri — Esri, Earthstar Geographics', maxZoom: 19
  }
);
var thunderforestOutdoorWc = L.tileLayer(
  'https://tile.thunderforest.com/outdoors/{z}/{x}/{y}.png?apikey=ff9168177a0e457dbaad796b08712512', {
    attribution: 'Maps © Thunderforest, Data © OSM contributors', maxZoom: 22
  }
);
var thunderforestAtlasWc = L.tileLayer(
  'https://tile.thunderforest.com/atlas/{z}/{x}/{y}.png?apikey=ff9168177a0e457dbaad796b08712512', {
    attribution: 'Maps © Thunderforest', maxZoom: 22
  }
);
var thunderforestLandscapeWc = L.tileLayer(
  'https://tile.thunderforest.com/landscape/{z}/{x}/{y}.png?apikey=ff9168177a0e457dbaad796b08712512', {
    attribution: 'Maps © Thunderforest', maxZoom: 22
  }
);
var stadiaAlidadeWc = L.tileLayer(
  'https://tiles.stadiamaps.com/tiles/alidade_smooth_dark/{z}/{x}/{y}.png?api_key=a8a97fb6-0fdd-4f58-a626-8f8b8f7850d6', {
    attribution: '© Stadia Maps, © OpenMapTiles, © OSM contributors', maxZoom: 20
  }
);
var stadiaOutWc = L.tileLayer(
  'https://tiles.stadiamaps.com/tiles/outdoors/{z}/{x}/{y}.png?api_key=a8a97fb6-0fdd-4f58-a626-8f8b8f7850d6', {
    attribution: '© Stadia Maps, © OpenMapTiles', maxZoom: 20
  }
);
var cartoDBWc = L.tileLayer(
  'https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
    attribution: '&copy; CARTO', maxZoom: 19
  }
);
var osmBrightWc = L.tileLayer(
  'https://tiles.stadiamaps.com/tiles/osm_bright/{z}/{x}/{y}.png?api_key=a8a97fb6-0fdd-4f58-a626-8f8b8f7850d6', {
    attribution: '© Stadia Maps, © OSM contributors', maxZoom: 20
  }
);

// 3) Add your chosen default layer to the detail map
thunderforestLandscapeWc.addTo(mapWcac);

// 4) Place the custom surveyor icon marker at the site center
var surveyorIconWcac = L.icon({
  iconUrl: '/assets/images/common/surveyor-icon.png',
  iconSize: [32,32],
  iconAnchor: [16,16],
  className: 'no-default-icon'
});
L.marker([37.218408, -107.847949], { icon: surveyorIconWcac })
  .addTo(mapWcac)
  .bindTooltip(
    "PUEBLO I COMMUNITY ON BLUE MESA, CO",
    { permanent: true, direction: 'bottom', className: 'custom-label' }
  );
  
  // ─────────────────────────────────────────────────────────────
//  BLUE MESA OVERVIEW MAP (small circle in the right pane)
// ─────────────────────────────────────────────────────────────

// 5) Initialize the overview map with all interactions off
var mapWcover = L.map('map-wcac-overview', {
  zoomControl: false,
  dragging: false,
  scrollWheelZoom: false,
  doubleClickZoom: false,
  boxZoom: false,
  keyboard: false,
  tap: false,
  touchZoom: false
})
  .setView([37.218408, -107.847949], 3);  // zoomed-out view

// 6) Add a simple gray‐canvas basemap for the overview
L.tileLayer(
  'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Light_Gray_Base/MapServer/tile/{z}/{y}/{x}', {
    attribution: 'Tiles © Esri — Esri, DeLorme, NAVTEQ', maxZoom: 16
  }
).addTo(mapWcover);

// 7) Plot a circle marker showing the project area in the overview
L.circleMarker([37.218408, -107.847949], {
  radius: 4,
  color: '#a32638',
  fillColor: '#a32638',
  fillOpacity: 1
})
  .addTo(mapWcover)
  .bindTooltip(
    "PROJECT AREA",
    { permanent: true, direction: 'top', className: 'custom-label2' }
  );

// ─────────────────────────────────────────────────────────────
//  LAYERS CONTROL for the detail map
// ─────────────────────────────────────────────────────────────
L.control.layers(
  {
    "🌳 Thunderforest Landscape": thunderforestLandscapeWc,
    "🛰️ Esri Satellite": esriWorldImageryWc,
    "🟩 Stamen Terrain": stamenTerrainWc,
    "🗺️ Thunderforest Atlas": thunderforestAtlasWc,
	"🏞️ Thunderforest Outdoors": thunderforestOutdoorWc,
    "🟪 Stadia Dark Alidade": stadiaAlidadeWc,
    "🟦 Stadia Outdoors": stadiaOutWc,
    "🟫 CartoDB Positron": cartoDBWc,
    "⬜️ OSM Bright": osmBrightWc
  },
  null,                // no overlays for BLUE MEsa
  { position: 'topleft', collapsed: false }
).addTo(mapWcac);
